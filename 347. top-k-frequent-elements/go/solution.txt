func topKFrequent(nums []int, k int) []int {
    freqMap := make(map[int]int)
    for _, n := range nums {
        freqMap[n]++
    }

    bucket := make([][]int, len(nums)+1)

    for k, v := range freqMap {
        bucket[v] = append(bucket[v], k)
    }

    var ans []int
    for i := len(bucket)-1; i >= 0; i-- {
        ans = append(ans, bucket[i]...)

        if len(ans) >= k {
            break
        }
    }
    return ans
}
