/**
 * Definition for singly-linked list.
 * type ListNode struct {
 *     Val int
 *     Next *ListNode
 * }
 */
func sortList(head *ListNode) *ListNode {
    // 0, 1 element, no need to sort, base case
    if head == nil || head.Next == nil {
        return head
    }

    // find middle 
    var prev *ListNode
    slow, fast := head, head

    for fast != nil && fast.Next != nil {
        prev = slow
        slow = slow.Next
        fast = fast.Next.Next
    }

    // cut off the first half
    prev.Next = nil

    left := sortList(head)
    right := sortList(slow)

    return merge(left, right)
}

func merge(l1 *ListNode, l2 *ListNode) *ListNode {
    dummy := &ListNode{}
    cur := dummy

    for l1 != nil && l2 != nil {
        if l1.Val < l2.Val {
            cur.Next = l1
            l1 = l1.Next
        } else {
            cur.Next = l2
            l2 = l2.Next
        }

        cur = cur.Next
    }

    // apend the rest
    if l1 == nil {
        cur.Next = l2
    }

    // ap[pend the rest]
    if l2 == nil {
        cur.Next = l1
    }

    return dummy.Next
}
