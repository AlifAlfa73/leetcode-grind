// reverse number solution
func isPalindrome(x int) bool {
    if  x < 0 {
        return false
    }

    reverse := 0
    origX := x
    for x > 0 {
        reverse *= 10
        digit := x % 10
        reverse += digit
        x = x / 10
    }

    return reverse == origX
}
