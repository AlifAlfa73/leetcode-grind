/**
 * Definition for singly-linked list.
 * type ListNode struct {
 *     Val int
 *     Next *ListNode
 * }
 */
func addTwoNumbers(l1 *ListNode, l2 *ListNode) *ListNode {
    dummyHead := &ListNode{
        Val : 0,
        Next: nil,
    }
    curAns := dummyHead
    carry := 0
    for l1 != nil || l2 != nil {
        dig1 := 0
        dig2 := 0
        if l1 != nil {
            dig1 = l1.Val
            l1 = l1.Next
        }

        if l2 != nil {
            dig2 = l2.Val
            l2 = l2.Next
        }

        ans := carry + dig1 + dig2
        carry = ans / 10
        ans %= 10

        ansNode := &ListNode{
            Val : ans,
            Next: nil,
        }

        curAns.Next = ansNode
        curAns = curAns.Next

        // fmt.Println(ans)
    }

    if carry != 0 {
        curAns.Next = &ListNode{
            Val : 1,
            Next: nil,
        }
    }

 

    return dummyHead.Next
}
