func longestCommonPrefix(strs []string) string {
    prefix := strs[0]

    for i := 1; i < len(strs); i++ {
        str := strs[i]
        commonPrefixLength := min(len(prefix), len(str))
        for commonPrefixLength > 0 {
            if prefix[:commonPrefixLength] == str[:commonPrefixLength] {
                break
            }

            commonPrefixLength--
        }

        prefix = prefix[:commonPrefixLength]
    }

    return prefix
}

func min(a, b int) int {
    if a < b {
        return a
    }

    return b
}
