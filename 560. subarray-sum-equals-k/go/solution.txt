func subarraySum(nums []int, k int) int {
    var kCount int
    prefixSum := make([]int, len(nums))
    sum := 0
    for i := 0; i < len(nums); i++ {
        sum += nums[i]
        prefixSum[i] = sum

        if sum == k {
            kCount++
        }
    }

    // calculate sum for each subscontiguous
    for i := 0; i < len(nums); i++ {
        for j:= i + 1; j < len(nums); j++ {
            // sum i, j
            kSum := prefixSum[j] - prefixSum[i]
            if kSum == k {
                kCount++
            }
        }
    }

    fmt.Println(prefixSum)
    return kCount
}
