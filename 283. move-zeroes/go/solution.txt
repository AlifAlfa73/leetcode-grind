func moveZeroes(nums []int)  {
    lastNonZeroElementIdx := 0

    for i := 0; i < len(nums); i++ {
        if nums[i] != 0 {
            temp := nums[i]
            nums[i] = nums[lastNonZeroElementIdx]
            nums[lastNonZeroElementIdx] = temp
            lastNonZeroElementIdx++
        }
    }
}
