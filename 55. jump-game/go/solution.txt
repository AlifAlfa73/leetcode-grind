func canJump(nums []int) bool {

    maxPoint := nums[0]

    for i := 1; i <= maxPoint && i < len(nums); i++ {
        if i + nums[i] > maxPoint {
            maxPoint = i + nums[i]
        }
    }

    return maxPoint >= len(nums) - 1
